void gravarPadrao ( int n )

  { printf ( "Erro ao abrir arquivo.\n" ); }

    { int y = n - 1;                       // while decrescente preenche a linha
      while ( y >= 0 )
      { if ( y >= x ) { linha[y] = x + 1; }
        else          { linha[y] = 0; }

    }
    fclose ( arq );
  }
}
